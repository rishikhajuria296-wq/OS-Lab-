PRACTICAL 01: GET PROCESS ID USING getpid()

AIM: To write a C program to display the Process ID (PID) of the currently executing process using getpid().

SOURCE CODE:

#include <stdio.h> #include <unistd.h>

int main() { printf("Current PID:%d\n", getpid()); return 0; }

PRACTICAL 02: PROCESS CREATION USING fork() AND wait()

AIM: To create a child process using fork() and make the parent process wait for the child to terminate using wait().

SOURCE CODE:

#include <stdio.h> #include <stdlib.h> #include <unistd.h> #include <sys/wait.h>

int main() { pid_t p = fork();

if(p == 0) {
    printf("Child Running with PID:%d\n", getpid());
    sleep(0);
    printf("Child Exiting...\n");
    exit(0);
}
else {
    printf("Parent Waiting for child to terminate...\n");
    pid_t terminated_pid = wait(NULL);
    printf("Parent Terminated Child PID:%d\n", terminated_pid);
}

return 0;


}

PRACTICAL 03: CHILD PROCESS TERMINATION AND REAPING

AIM: To create a child process using fork() and use wait() to ensure that the parent process collects the child's termination status and reaps the child process.

SOURCE CODE:

#include <stdio.h> #include <sys/wait.h> #include <stdlib.h> #include <unistd.h>

int main() { pid_t p = fork();

if(p == 0) {
    printf("Child working...\n");
    exit(0);
}
else {
    pid_t child_pid = wait(NULL);
    printf("Child %d reaped successfully!\n", child_pid);
}

return 0;


}
